%%[
/* ============================================================================
   API CREDENTIAL ENCRYPTION TOOL — CloudPage
   PLAIN-ENGLISH COMMENTED VERSION — every step explains what it does and why,
   written for a non-technical reader.

   The flow this page walks someone through:
   Enter Email -> Verify against Salesforce -> Choose Add/Update ->
   Enter Details -> Save (encrypted) -> Success
   ============================================================================ */

  /* Setting up empty "boxes" to hold information we'll need further down.
     Nothing happens yet — this just gives each piece of information a name. */
  VAR @EmailAddress, @isVerifyAmp, @isSubmitAmp, @isValidUser, @rs, @rowCount, @activeFlag, @PageURL
  VAR @Mode, @EncClientID, @EncClientSecret, @EncAuthBaseURI, @EncRestBaseURI, @EncNewSecret, @null

  /* Remember this page's own web address, so we can send the browser back
     to "this same page" after a successful save. */
  SET @null = ""
  SET @PageURL = "https://cloud.email.kpmgus.com/API_Credential_Encryption_Tool_DEV"

  /* Read whatever the user typed into the Email box, trimming stray spaces. */
  SET @EmailAddress = Trim(RequestParameter('txtEmailAddress'))

  /* Was the "Verify & Continue" button clicked? */
  SET @isVerifyAmp = RequestParameter('btnVerify')

  /* Was the "Save Securely" button clicked? */
  SET @isSubmitAmp = RequestParameter('btnSubmit')

  /* Which action did the user choose: "add" a new package or "update" one? */
  SET @Mode = RequestParameter('ddlMode')

  /* Start by assuming the person is NOT verified. This only flips to 'true'
     further down if the Salesforce check actually passes. */
  SET @isValidUser = 'false'

  /* --- THE SECURITY CHECK -----------------------------------------------
     We check the email against Salesforce whenever EITHER the Verify button
     OR the final Save button was clicked (not just once) — this way, even
     if someone tampered with the page to skip straight to Save, we still
     verify their email right before actually saving anything. */
  IF NOT EMPTY(@EmailAddress) AND (@isVerifyAmp == 'Verify' OR @isSubmitAmp == 'Submit') THEN

    /* Ask Salesforce: "Do you have an active User with this email?" */
    SET @rs = RetrieveSalesforceObjects("User", "Id,Username,Email,IsActive", "Email", "=", @EmailAddress)

    /* Count how many matches came back (should be 0 or 1). */
    SET @rowCount = RowCount(@rs)

    IF @rowCount > 0 THEN
      /* Found someone — check if their Salesforce account is active. */
      SET @activeFlag = Field(Row(@rs,1),"IsActive")

      IF @activeFlag == "true" OR @activeFlag == "1" THEN
        /* Confirmed: real, active Salesforce user. Allowed to continue. */
        SET @isValidUser = 'true'
      ENDIF
    ENDIF
  ENDIF

  /* --- ENCRYPTING THE SENSITIVE INFORMATION ------------------------------
     Only runs if the Save button was clicked AND the security check above
     passed. If either is untrue, nothing here gets encrypted or saved. */
  IF @isSubmitAmp == 'Submit' AND @isValidUser == 'true' THEN

    IF @Mode == 'add' THEN
      /* Scramble (encrypt) each sensitive value using the same method/
         password your other Marketing Cloud pages already use — nothing
         new is being introduced, just reused consistently. */
      SET @EncClientID     = EncryptSymmetric(RequestParameter('txtClientID'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
      SET @EncClientSecret = EncryptSymmetric(RequestParameter('txtClientSecret'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
      SET @EncAuthBaseURI  = EncryptSymmetric(RequestParameter('txtAuthBaseURI'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
      SET @EncRestBaseURI  = EncryptSymmetric(RequestParameter('txtRESTBaseURI'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
    ELSEIF @Mode == 'update' THEN
      /* Only the one new secret needs scrambling for an update. */
      SET @EncNewSecret    = EncryptSymmetric(RequestParameter('txtNewClientSecret'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
    ENDIF
  ENDIF
]%%

<script runat="server">
/* ============================================================================
   Everything below is the "business logic" — the rules for what should
   happen next (save it, reject it, show an error, which screen to display).
   ============================================================================ */

Platform.Load("Core", "1.1.1"); // Turns on the toolbox needed to talk to Data Extensions.

// Tell browsers/proxies/SFMC's CDN: never save a copy of this page, always
// fetch fresh. This page shows information specific to one person's one
// visit — we don't want a cached copy accidentally shown to someone else.
Platform.Response.SetResponseHeader("Cache-Control", "no-store, no-cache, must-revalidate, max-age=0");
Platform.Response.SetResponseHeader("Pragma", "no-cache");
Platform.Response.SetResponseHeader("Expires", "0");

var deDetails = "ENT.API_Details_GEE-61699";          // The existing "filing cabinet" where package credentials live.
var deAudit   = "ENT.API_Credential_Audit_Log"; // The new "filing cabinet" that logs every attempt.
var pageUrl   = Variable.GetValue("@PageURL"); // This page's own address, from the AMPscript section above.

var emailAddress = Variable.GetValue("@EmailAddress"); // The email the user typed.
var isValidUser  = (Variable.GetValue("@isValidUser") == "true"); // Did it pass the Salesforce check?
var isVerify     = Request.GetFormField("btnVerify"); // Was "Verify & Continue" clicked?
var isSubmit     = Request.GetFormField("btnSubmit"); // Was "Save Securely" clicked?
var mode         = Request.GetFormField("ddlMode");   // "add" or "update".

// The already-scrambled (encrypted) versions of the sensitive values,
// calculated earlier in the AMPscript section above.
var encClientId      = Variable.GetValue("@EncClientID");
var encClientSecret  = Variable.GetValue("@EncClientSecret");
var encAuthBaseURI   = Variable.GetValue("@EncAuthBaseURI");
var encRestBaseURI   = Variable.GetValue("@EncRestBaseURI");
var encNewSecret     = Variable.GetValue("@EncNewSecret");

var successMsg  = ""; // Will hold a friendly "it worked!" message.
var errorMsg    = ""; // Will hold a friendly error message.
var renderStep  = "email"; // Which of the page's screens to show: email | choose | add | update | done.
                           // Defaults to "email" — the very first screen — unless something below changes it.

// ---- HANDLING THE "AFTER A SUCCESSFUL SAVE" PAGE LOAD ----------------------
// After a successful Add or Update further down, we deliberately send the
// browser to reload this same page fresh, carrying a short "it worked" note
// in the web address rather than showing success directly. This section
// reads that note back. Benefit: if the user hits Refresh, they just reload
// this plain "it worked" page again — nothing gets accidentally saved twice.
var qsStatus = Request.GetQueryStringParameter("status"); // Did we arrive here right after a success?
var qsAction = Request.GetQueryStringParameter("action");  // Was it "add" or "update"?
var qsPkg    = Request.GetQueryStringParameter("pkg");     // Which package name?

if (qsStatus == "success") {
  renderStep = "done"; // Show the final success screen.
  if (qsAction == "add") {
    successMsg = "Package '" + qsPkg + "' was created successfully. All sensitive values were AES-encrypted before storage.";
  } else if (qsAction == "update") {
    successMsg = "Client secret for '" + qsPkg + "' was updated successfully.";
  }
}

// If the user's last attempt failed validation, we want to put whatever
// they typed (except secrets) back into the boxes so they don't have to
// retype everything. These lines just re-read what was submitted.
var f_packageName      = Trim(Request.GetFormField("txtPackageName"));
var f_authBaseURI       = Trim(Request.GetFormField("txtAuthBaseURI"));
var f_restBaseURI       = Trim(Request.GetFormField("txtRESTBaseURI"));
var f_expirationDate    = Trim(Request.GetFormField("txtExpirationDate"));
var f_businessUnit      = Trim(Request.GetFormField("txtBusinessUnit"));
var f_scope             = Trim(Request.GetFormField("txtScope"));
var f_packageNameUpdate = Trim(Request.GetFormField("txtPackageNameUpdate"));
var f_newExpiration     = Trim(Request.GetFormField("txtNewExpirationDate"));

// A reusable helper: whenever we need to write one entry into the audit log
// (the "who did what, when" history), we call this instead of repeating the
// same lines every time. Wrapped in a safety net so a logging hiccup never
// crashes the whole page.
function logAudit(actionType, pkgName, bu, who, status, errMsg) {
  try {
    Platform.Function.InsertData(
      deAudit,
      ["AuditID","ActionType","PackageName","BusinessUnit","PerformedBy","ActionDate","Status","ErrorMessage"],
      [Platform.Function.GUID(), actionType, pkgName, bu, who, Now(), status, errMsg]
    );
  } catch (logEx) { /* if logging itself fails, fail quietly rather than alarming the user */ }
}

// ---- SCENARIO 1: The user just clicked "Verify & Continue" ---------------
if (isVerify == "Verify") {

  if (Length(emailAddress) == 0) {
    errorMsg = "Please enter your email address.";
    renderStep = "email"; // Stay on the email screen.
  } else if (!isValidUser) {
    errorMsg = "We couldn't verify '" + emailAddress + "' as an active Salesforce user. Please check the address and try again.";
    logAudit("Access Denied", "", "", emailAddress, "Failed", "Email not found or inactive in Salesforce User object");
    renderStep = "email"; // Still stay on the email screen — they need to try again.
  } else {
    renderStep = "choose"; // Success — move them on to the "choose an action" screen.
  }

// ---- SCENARIO 2: The user just clicked "Save Securely" --------------------
} else if (isSubmit == "Submit") {

  // Defense-in-depth: even on the final Save, we double-check the email is
  // still verified (in case someone tried to skip ahead or tamper with the
  // page). If not, send them back to square one.
  if (Length(emailAddress) == 0 || !isValidUser) {

    errorMsg = "Your verification could not be confirmed. Please re-enter your email address and verify again.";
    renderStep = "email";
    if (Length(emailAddress) > 0) {
      logAudit("Access Denied", "", "", emailAddress, "Failed", "Re-verification failed on final submit");
    }

  // ---- PATH A: Adding a new package ---------------------------------------
  } else if (mode == "add") {

    renderStep = "add"; // If anything below fails, stay on the Add screen (with the error shown).
    var packageName  = f_packageName;
    var clientId     = Trim(Request.GetFormField("txtClientID"));
    var clientSecret = Trim(Request.GetFormField("txtClientSecret"));
    var authBaseURI  = f_authBaseURI;
    var restBaseURI  = f_restBaseURI;
    var expirationDt = f_expirationDate;
    var businessUnit = f_businessUnit;
    var scope        = f_scope;

    // Make sure nothing important was left blank.
    if (Length(packageName) == 0 || Length(clientId) == 0 || Length(clientSecret) == 0 ||
        Length(authBaseURI) == 0 || Length(restBaseURI) == 0 || Length(expirationDt) == 0 ||
        Length(businessUnit) == 0) {

      errorMsg = "All fields are required to add a new package.";

    } else {

      try {

        // Check first: does a package with this name already exist? Avoid
        // accidentally creating a duplicate.
        var existing = Platform.Function.Lookup(deDetails, "API Name", "API Name", packageName);

        if (existing != null && String(existing).length > 0) {

          errorMsg = "A package named '" + packageName + "' already exists. Please use Update instead.";
          logAudit("Create", packageName, businessUnit, emailAddress, "Failed", "Duplicate package name");

        } else {

          // Save the new record. Notice the sensitive fields being saved
          // (encClientId, encClientSecret, etc.) are the SCRAMBLED versions
          // calculated earlier — the plain-text version the user typed is
          // never written to storage.
          Platform.Function.InsertData(
            deDetails,
            ["API Name","Encrypted Client Id","Encrypted Client Secret","Encrypted Authentication Base URI",
             "Encrypted REST Base URI","Business Unit","Scope","Client Secret Expiration Date","Last Modified Date"],
            [packageName, encClientId, encClientSecret, encAuthBaseURI,
             encRestBaseURI, businessUnit, scope, expirationDt, Now()]
          );

          logAudit("Create", packageName, businessUnit, emailAddress, "Success", "");

          // Instead of showing "Success" directly on this same page load,
          // redirect the browser to a fresh page load (see the "AFTER A
          // SUCCESSFUL SAVE" section above). This is what prevents someone
          // accidentally creating the same package twice via browser Refresh.
          var redirectUrlAdd = pageUrl + "?status=success&action=add&pkg=" + encodeURIComponent(packageName);
          Platform.Response.Redirect(redirectUrlAdd, true);
        }

      } catch (addEx) {
        // Something unexpected went wrong technically — show a friendly
        // message, but log the real technical detail for support purposes.
        errorMsg = "An unexpected error occurred while creating the package. Please contact your Marketing Cloud administrator.";
        logAudit("Create", packageName, businessUnit, emailAddress, "Failed", Stringify(addEx).substring(0, 3900));
      }
    }

  // ---- PATH B: Updating an existing package -------------------------------
  } else if (mode == "update") {

    renderStep = "update"; // If anything below fails, stay on the Update screen.
    var packageName2  = f_packageNameUpdate;
    var newSecret     = Trim(Request.GetFormField("txtNewClientSecret"));
    var newExpiration = f_newExpiration;

    if (Length(packageName2) == 0 || Length(newSecret) == 0 || Length(newExpiration) == 0) {

      errorMsg = "Package Name, New Client Secret, and New Expiration Date are all required.";

    } else {

      try {

        // Confirm the package actually exists before trying to update it.
        var rows = Platform.Function.LookupRows(deDetails, "API Name", packageName2);

        if (rows == null || rows.length == 0) {

          errorMsg = "No package found named '" + packageName2 + "'. Please check the Package Name and try again.";
          logAudit("Update", packageName2, "", emailAddress, "Failed", "Package not found");

        } else {

          // Just grabbing the existing Business Unit so we can include it
          // in the audit log entry — it isn't being changed.
          var businessUnit2 = rows[0].Client.Values["Business Unit"];

          // Update ONLY the three things that should change: the new
          // (scrambled) secret, its expiration date, and today's date as
          // "last modified." Everything else on this record is deliberately
          // left untouched.
          Platform.Function.UpdateData(
            deDetails,
            ["API Name"], [packageName2],
            ["Encrypted Client Secret", "Client Secret Expiration Date", "Last Modified Date"],
            [encNewSecret, newExpiration, Now()]
          );

          logAudit("Update", packageName2, businessUnit2, emailAddress, "Success", "");

          // Same idea as the Add path: redirect to a fresh page load instead
          // of showing success directly, so a Refresh can't redo this update.
          var redirectUrlUpd = pageUrl + "?status=success&action=update&pkg=" + encodeURIComponent(packageName2);
          Platform.Response.Redirect(redirectUrlUpd, true);
        }

      } catch (updEx) {
        errorMsg = "An unexpected error occurred while updating the package. Please contact your Marketing Cloud administrator.";
        logAudit("Update", packageName2, "", emailAddress, "Failed", Stringify(updEx).substring(0, 3900));
      }
    }

  } else {
    // Save was clicked but somehow no valid action was chosen — send them
    // back to the "choose an action" screen.
    renderStep = "choose";
  }
}

// Everything calculated above (which screen to show, any messages, what the
// user typed) needs to be handed over to the HTML/design section below.
// These lines are that hand-off.
Variable.SetValue("@renderStep", renderStep);
Variable.SetValue("@successMsg", successMsg);
Variable.SetValue("@errorMsg", errorMsg);
Variable.SetValue("@mode", mode);
Variable.SetValue("@emailAddress", emailAddress);
Variable.SetValue("@f_packageName", f_packageName);
Variable.SetValue("@f_authBaseURI", f_authBaseURI);
Variable.SetValue("@f_restBaseURI", f_restBaseURI);
Variable.SetValue("@f_expirationDate", f_expirationDate);
Variable.SetValue("@f_businessUnit", f_businessUnit);
Variable.SetValue("@f_scope", f_scope);
Variable.SetValue("@f_packageNameUpdate", f_packageNameUpdate);
Variable.SetValue("@f_newExpiration", f_newExpiration);
</script>

<script>
  /* This is the JavaScript that runs IN THE PERSON'S BROWSER (as opposed to
     the server-side logic near the top of this file). It's what makes
     switching between screens feel instant, with no page reload, whenever
     that switch doesn't actually need to talk to the server. */

  // Hides every screen, then shows just the one requested, and updates the
  // "1-2-3" progress stepper to match. Also hides the stepper entirely once
  // we're on the final "done" screen (there's nothing left to track progress
  // toward at that point).
  function showScreen(name, stepperStep) {
    document.querySelectorAll('.screen').forEach(function(el) { el.classList.remove('active'); });
    document.getElementById('screen-' + name).classList.add('active');
    document.getElementById('stepper').setAttribute('data-active', stepperStep);
    document.getElementById('stepper').style.display = (name === 'done') ? 'none' : 'flex';
  }

  // Runs when someone clicks either the "Add New Package" or "Update
  // Existing Package" card. Records their choice in the hidden field, then
  // switches to the matching Details screen — no server round-trip needed.
  function chooseMode(mode) {
    document.getElementById('ddlMode').value = mode;
    showScreen(mode, '3');
  }

  // Runs when someone clicks "Back" from a Details screen. Clears the
  // hidden mode field and returns them to the "Choose an action" screen.
  function backToChoose() {
    document.getElementById('ddlMode').value = '';
    showScreen('choose', '2');
  }

  // Runs when someone clicks "Do Another" on the success screen. Reloads
  // the page at its plain web address (with no leftover details attached),
  // which puts them back at square one — a completely fresh start.
  function startOver() {
    window.location.href = window.location.href.split('?')[0];
  }

  /* When the page first loads (or reloads after a submission), this decides
     which screen to actually show, based on what the server-side logic
     above decided ("renderStep"). This is what makes sure that, for
     example, a failed submission on the Add screen brings the user back to
     the Add screen (with their entries and the error message still there)
     instead of dumping them back to the very first screen. */
  var serverStep = "%%=v(@renderStep)=%%";
  var serverMode = "%%=v(@mode)=%%";

  if (serverStep === "choose") {
    showScreen('choose', '2');
  } else if (serverStep === "add") {
    document.getElementById('ddlMode').value = 'add';
    showScreen('add', '3');
  } else if (serverStep === "update") {
    document.getElementById('ddlMode').value = 'update';
    showScreen('update', '3');
  } else if (serverStep === "done") {
    showScreen('done', '3');
  } else {
    // Default: nothing special happened yet (a brand new visit), so start
    // at the very first screen.
    showScreen('email', '1');
  }
</script>
