   Flow: Enter Email -> Verify against Salesforce -> Choose Add/Update ->
         Enter Details -> Save (encrypted) -> Success
   ============================================================================ */

  /* ---- FIX 1 --------------------------------------------------------------
     EVERY variable that SSJS later writes with Variable.SetValue(), and every
     variable printed with %%=v(@x)=%% further down, MUST be declared here.
     Missing declarations were the main reason the page died on load. */
  VAR @EmailAddress, @isVerifyAmp, @isSubmitAmp, @isValidUser, @rs, @rowCount, @activeFlag, @PageURL
  VAR @Mode, @EncClientID, @EncClientSecret, @EncAuthBaseURI, @EncRestBaseURI, @EncNewSecret, @null
  VAR @renderStep, @successMsg, @errorMsg
  VAR @f_packageName, @f_authBaseURI, @f_restBaseURI, @f_expirationDate
  VAR @f_businessUnit, @f_scope, @f_packageNameUpdate, @f_newExpiration

  /* Give everything a safe starting value so nothing is ever "undefined"
     when the HTML below tries to print it. */
  SET @null            = ""
  SET @renderStep      = "email"
  SET @successMsg      = ""
  SET @errorMsg        = ""
  SET @EncClientID     = ""
  SET @EncClientSecret = ""
  SET @EncAuthBaseURI  = ""
  SET @EncRestBaseURI  = ""
  SET @EncNewSecret    = ""
  SET @f_packageName       = ""
  SET @f_authBaseURI       = ""
  SET @f_restBaseURI       = ""
  SET @f_expirationDate    = ""
  SET @f_businessUnit      = ""
  SET @f_scope             = ""
  SET @f_packageNameUpdate = ""
  SET @f_newExpiration     = ""

  /* This page's own address — used for the form action and the post-save
     redirect. Must exactly match the published CloudPage URL. */
  SET @PageURL = "https://cloud.email.kpmgus.com/API_Credential_Encryption_Tool_DEV"

  /* Read what came back from the form. */
  SET @EmailAddress = Trim(RequestParameter('txtEmailAddress'))
  SET @isVerifyAmp  = RequestParameter('btnVerify')
  SET @isSubmitAmp  = RequestParameter('btnSubmit')
  SET @Mode         = RequestParameter('ddlMode')

  /* Assume NOT verified until the Salesforce check below passes. */
  SET @isValidUser = 'false'

  /* --- THE SECURITY CHECK -----------------------------------------------
     Runs on Verify AND on the final Save, so tampering with the page to skip
     straight to Save still gets checked. */
  IF NOT EMPTY(@EmailAddress) AND (@isVerifyAmp == 'Verify' OR @isSubmitAmp == 'Submit') THEN

    SET @rs = RetrieveSalesforceObjects("User", "Id,Username,Email,IsActive", "Email", "=", @EmailAddress)
    SET @rowCount = RowCount(@rs)

    IF @rowCount > 0 THEN
      SET @activeFlag = Field(Row(@rs,1),"IsActive")

      IF @activeFlag == "true" OR @activeFlag == "True" OR @activeFlag == "1" THEN
        SET @isValidUser = 'true'
      ENDIF
    ENDIF
  ENDIF

  /* --- ENCRYPTION ---------------------------------------------------------
     EncryptSymmetric is AMPscript-only, so it has to happen here, before the
     SSJS block. Only runs when Save was clicked AND verification passed.

     SECURITY NOTE: the password / salt / IV below are hard-coded literals.
     Anyone who can open this CloudPage in Content Builder can decrypt every
     stored credential. Before production, create Key Management entries and
     pass their external keys in the 3rd, 5th and 7th arguments instead of
     @null, e.g.:
       EncryptSymmetric(@val,"AES","APIToolPwdKey",@null,"APIToolSaltKey",@null,"APIToolIVKey",@null)
  */
  IF @isSubmitAmp == 'Submit' AND @isValidUser == 'true' THEN

    IF @Mode == 'add' THEN
      SET @EncClientID     = EncryptSymmetric(RequestParameter('txtClientID'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
      SET @EncClientSecret = EncryptSymmetric(RequestParameter('txtClientSecret'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
      SET @EncAuthBaseURI  = EncryptSymmetric(RequestParameter('txtAuthBaseURI'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
      SET @EncRestBaseURI  = EncryptSymmetric(RequestParameter('txtRESTBaseURI'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
    ELSEIF @Mode == 'update' THEN
      SET @EncNewSecret    = EncryptSymmetric(RequestParameter('txtNewClientSecret'), "AES", @null, "password", @null, "0000000000000000", @null, "00000000000000000000000000000000")
    ENDIF
  ENDIF

<script runat="server">
Platform.Load("Core", "1.1.1");

/* ============================================================================
   SERVER-SIDE BUSINESS LOGIC

   IMPORTANT STRUCTURAL RULE:
   Every function below is declared at the TOP LEVEL of this block, never
   inside a try/catch or an if/else. SFMC's SSJS engine throws
   "Unable to retrieve security descriptor for this frame" when a function is
   declared inside a block and then called from nested code. Keep them here.
   ============================================================================ */

/* ---- DEBUG SWITCH -------------------------------------------------------
   true  = print the real error (plus the step it failed on) onto the page.
   false = show a friendly message. MUST be false before go-live. */
var DEBUG = true;

/* ---- CONFIG -------------------------------------------------------------
   The "ENT." prefix is ONLY correct if these are shared / enterprise DEs.
   If they live in the local Business Unit, remove the prefix. */
var deDetails = "ENT.API_Details_GEE-61699";
var deAudit   = "ENT.API_Credential_Audit_Log";

/* If your DE still has Client Id / Client Secret / Authentication Base URI
   marked as Nullable = No, we must write something into them or the insert
   is rejected. We write a placeholder, never the real value.
   RECOMMENDED: mark those three columns nullable and set this to false. */
var deHasRequiredPlainColumns = true;

/* ---- BREADCRUMB ---------------------------------------------------------
   Updated as we move through the logic. If anything throws, DEBUG prints
   this alongside the error so you know exactly where it died. */
var dbgStep = "start";

/* ============================ HELPER FUNCTIONS ============================ */

/* Read a posted form field safely. GetFormField returns null on a plain GET,
   and Platform.Function.Trim(null) throws, so everything goes through here. */
function ff(name) {
  var v = Request.GetFormField(name);
  if (v == null) { return ""; }
  return Platform.Function.Trim(String(v));
}

/* Read a query string value safely. */
function qs(name) {
  var v = Request.GetQueryStringParameter(name);
  if (v == null) { return ""; }
  return String(v);
}

/* Length of anything, null-safe. There is no global Length() in SSJS. */
function len(v) {
  if (v == null) { return 0; }
  return String(v).length;
}

/* Escape values before they are printed back into HTML, so a stray < or "
   cannot break the markup. Uses split/join rather than regex — SSJS regex
   support is patchy and not worth the risk here. */
function esc(v) {
  if (v == null) { return ""; }
  var s = String(v);
  s = s.split("&").join("&amp;");
  s = s.split("<").join("&lt;");
  s = s.split(">").join("&gt;");
  s = s.split('"').join("&quot;");
  return s;
}

/* Write one row into the audit log. Never allowed to crash the page. */
function logAudit(actionType, pkgName, bu, who, status, errMsg) {
  try {
    Platform.Function.InsertData(
      deAudit,
      ["AuditID","ActionType","PackageName","BusinessUnit","PerformedBy","ActionDate","Status","ErrorMessage"],
      [Platform.Function.GUID(), actionType, pkgName, bu, who, Platform.Function.Now(), status, errMsg]
    );
  } catch (logEx) { /* logging must never break the user's journey */ }
}

/* Insert a brand new package record. Only encrypted values are stored. */
function insertPackage(packageName, encId, encSecret, encAuth, encRest, bu, scope, expDt) {
  var cols = ["API Name",
              "Encrypted Client Id","Encrypted Client Secret",
              "Encrypted Authentication Base URI","Encrypted REST Base URI",
              "Business Unit","Scope",
              "Client Secret Expiration Date","Last Modified Date"];

  var vals = [packageName,
              encId, encSecret,
              encAuth, encRest,
              bu, scope,
              expDt, Platform.Function.Now()];

  /* Placeholders purely to satisfy "Nullable = No" on the plain-text columns.
     Built with push() rather than concat() for engine safety. */
  if (deHasRequiredPlainColumns) {
    cols.push("Client Id");
    cols.push("Client Secret");
    cols.push("Authentication Base URI");
    vals.push("[ENCRYPTED]");
    vals.push("[ENCRYPTED]");
    vals.push("[ENCRYPTED]");
  }

  Platform.Function.InsertData(deDetails, cols, vals);
}

/* Update only the rotated secret, its expiry, and the last-modified stamp. */
function updatePackageSecret(packageName, encSecret, newExpiration) {
  Platform.Function.UpdateData(
    deDetails,
    ["API Name"], [packageName],
    ["Encrypted Client Secret", "Client Secret Expiration Date", "Last Modified Date"],
    [encSecret, newExpiration, Platform.Function.Now()]
  );
}

/* ============================ PAGE STATE ============================ */

var successMsg  = "";
var errorMsg    = "";
var renderStep  = "email";   // email | choose | add | update | done
var redirectUrl = "";        // set on success, acted on at the very end

var pageUrl      = "";
var emailAddress = "";
var isValidUser  = false;
var isVerify     = "";
var isSubmit     = "";
var mode         = "";

var encClientId     = "";
var encClientSecret = "";
var encAuthBaseURI  = "";
var encRestBaseURI  = "";
var encNewSecret    = "";

var f_packageName       = "";
var f_authBaseURI       = "";
var f_restBaseURI       = "";
var f_expirationDate    = "";
var f_businessUnit      = "";
var f_scope             = "";
var f_packageNameUpdate = "";
var f_newExpiration     = "";

/* ============================ MAIN LOGIC ============================ */

try {

  dbgStep = "read-ampscript-vars";

  pageUrl      = Variable.GetValue("@PageURL");
  emailAddress = Variable.GetValue("@EmailAddress");
  if (emailAddress == null) { emailAddress = ""; }

  isValidUser = (Variable.GetValue("@isValidUser") == "true");

  encClientId     = Variable.GetValue("@EncClientID");
  encClientSecret = Variable.GetValue("@EncClientSecret");
  encAuthBaseURI  = Variable.GetValue("@EncAuthBaseURI");
  encRestBaseURI  = Variable.GetValue("@EncRestBaseURI");
  encNewSecret    = Variable.GetValue("@EncNewSecret");

  dbgStep = "read-form-fields";

  isVerify = ff("btnVerify");
  isSubmit = ff("btnSubmit");
  mode     = ff("ddlMode");

  f_packageName       = ff("txtPackageName");
  f_authBaseURI       = ff("txtAuthBaseURI");
  f_restBaseURI       = ff("txtRESTBaseURI");
  f_expirationDate    = ff("txtExpirationDate");
  f_businessUnit      = ff("txtBusinessUnit");
  f_scope             = ff("txtScope");
  f_packageNameUpdate = ff("txtPackageNameUpdate");
  f_newExpiration     = ff("txtNewExpirationDate");

  dbgStep = "read-query-string";

  // After a successful save we come back here with a note in the URL, so
  // pressing Refresh can never save the same thing twice.
  var qsStatus = qs("status");
  var qsAction = qs("action");
  var qsPkg    = qs("pkg");

  if (qsStatus == "success") {
    renderStep = "done";
    if (qsAction == "add") {
      successMsg = "Package '" + qsPkg + "' was created successfully. All sensitive values were AES-encrypted before storage.";
    } else if (qsAction == "update") {
      successMsg = "Client secret for '" + qsPkg + "' was updated successfully.";
    }
  }

  // ---- SCENARIO 1: "Verify & Continue" -----------------------------------
  if (isVerify == "Verify") {

    dbgStep = "verify-branch";

    if (len(emailAddress) == 0) {
      errorMsg = "Please enter your email address.";
      renderStep = "email";
    } else if (!isValidUser) {
      errorMsg = "We couldn't verify '" + emailAddress + "' as an active Salesforce user. Please check the address and try again.";
      logAudit("Access Denied", "", "", emailAddress, "Failed", "Email not found or inactive in Salesforce User object");
      renderStep = "email";
    } else {
      renderStep = "choose";
    }

  // ---- SCENARIO 2: "Save Securely" ---------------------------------------
  } else if (isSubmit == "Submit") {

    dbgStep = "submit-branch";

    // Defence in depth: re-check verification on the final save too.
    if (len(emailAddress) == 0 || !isValidUser) {

      errorMsg = "Your verification could not be confirmed. Please re-enter your email address and verify again.";
      renderStep = "email";
      if (len(emailAddress) > 0) {
        logAudit("Access Denied", "", "", emailAddress, "Failed", "Re-verification failed on final submit");
      }

    // ---- PATH A: Add a new package ---------------------------------------
    } else if (mode == "add") {

      dbgStep = "add-path";
      renderStep = "add";

      var packageName  = f_packageName;
      var clientId     = ff("txtClientID");
      var clientSecret = ff("txtClientSecret");
      var authBaseURI  = f_authBaseURI;
      var restBaseURI  = f_restBaseURI;
      var expirationDt = f_expirationDate;
      var businessUnit = f_businessUnit;
      var scope        = f_scope;

      if (len(packageName) == 0 || len(clientId) == 0 || len(clientSecret) == 0 ||
          len(authBaseURI) == 0 || len(restBaseURI) == 0 || len(expirationDt) == 0 ||
          len(businessUnit) == 0 || len(scope) == 0) {

        errorMsg = "All fields are required to add a new package.";

      } else {

        try {

          dbgStep = "add-duplicate-check";
          var existing = Platform.Function.Lookup(deDetails, "API Name", "API Name", packageName);

          if (existing != null && String(existing).length > 0) {

            errorMsg = "A package named '" + packageName + "' already exists. Please use Update instead.";
            logAudit("Create", packageName, businessUnit, emailAddress, "Failed", "Duplicate package name");

          } else {

            dbgStep = "add-insert";
            insertPackage(packageName, encClientId, encClientSecret, encAuthBaseURI,
                          encRestBaseURI, businessUnit, scope, expirationDt);

            logAudit("Create", packageName, businessUnit, emailAddress, "Success", "");

            // Remember the URL only. The actual redirect happens at the very
            // end, OUTSIDE every try/catch — Redirect() terminates the request
            // by throwing internally, and a catch would swallow it and report
            // a false failure on a save that actually worked.
            redirectUrl = pageUrl + "?status=success&action=add&pkg=" + encodeURIComponent(packageName);
          }

        } catch (addEx) {
          errorMsg = "An unexpected error occurred while creating the package. Please contact your Marketing Cloud administrator.";
          logAudit("Create", packageName, businessUnit, emailAddress, "Failed",
                   String(Platform.Function.Stringify(addEx)).substring(0, 3900));
        }
      }

    // ---- PATH B: Update an existing package ------------------------------
    } else if (mode == "update") {

      dbgStep = "update-path";
      renderStep = "update";

      var packageName2  = f_packageNameUpdate;
      var newSecret     = ff("txtNewClientSecret");
      var newExpiration = f_newExpiration;

      if (len(packageName2) == 0 || len(newSecret) == 0 || len(newExpiration) == 0) {

        errorMsg = "Package Name, New Client Secret, and New Expiration Date are all required.";

      } else {

        try {

          dbgStep = "update-lookup";
          var rows = Platform.Function.LookupRows(deDetails, "API Name", packageName2);

          if (rows == null || rows.length == 0) {

            errorMsg = "No package found named '" + packageName2 + "'. Please check the Package Name and try again.";
            logAudit("Update", packageName2, "", emailAddress, "Failed", "Package not found");

          } else {

            // LookupRows returns plain row objects — rows[0]["Field Name"].
            var businessUnit2 = rows[0]["Business Unit"];
            if (businessUnit2 == null) { businessUnit2 = ""; }

            dbgStep = "update-write";
            updatePackageSecret(packageName2, encNewSecret, newExpiration);

            logAudit("Update", packageName2, businessUnit2, emailAddress, "Success", "");

            redirectUrl = pageUrl + "?status=success&action=update&pkg=" + encodeURIComponent(packageName2);
          }

        } catch (updEx) {
          errorMsg = "An unexpected error occurred while updating the package. Please contact your Marketing Cloud administrator.";
          logAudit("Update", packageName2, "", emailAddress, "Failed",
                   String(Platform.Function.Stringify(updEx)).substring(0, 3900));
        }
      }

    } else {
      // Save clicked but no action chosen — back to the choose screen.
      renderStep = "choose";
    }
  }

} catch (pageEx) {

  renderStep = "email";
  if (DEBUG) {
    Write("<pre style='background:#FDECEC;color:#9A2626;padding:12px;font-size:12px;white-space:pre-wrap;'>"
          + "FAILED AT STEP: " + dbgStep + "\n"
          + Platform.Function.Stringify(pageEx) + "</pre>");
  } else {
    errorMsg = "A technical error occurred. Please contact your Marketing Cloud administrator.";
  }
}

/* ---- HAND OFF TO THE HTML BELOW ---------------------------------------
   Outside the try/catch, so the page always renders with valid values even
   if something above failed. */
Variable.SetValue("@renderStep",  renderStep);
Variable.SetValue("@successMsg",  esc(successMsg));
Variable.SetValue("@errorMsg",    esc(errorMsg));
Variable.SetValue("@mode",        esc(mode));
Variable.SetValue("@EmailAddress", esc(emailAddress));
Variable.SetValue("@f_packageName",       esc(f_packageName));
Variable.SetValue("@f_authBaseURI",       esc(f_authBaseURI));
Variable.SetValue("@f_restBaseURI",       esc(f_restBaseURI));
Variable.SetValue("@f_expirationDate",    esc(f_expirationDate));
Variable.SetValue("@f_businessUnit",      esc(f_businessUnit));
Variable.SetValue("@f_scope",             esc(f_scope));
Variable.SetValue("@f_packageNameUpdate", esc(f_packageNameUpdate));
Variable.SetValue("@f_newExpiration",     esc(f_newExpiration));

/* Now — and only now, clear of every try/catch — perform the redirect. */
if (redirectUrl != "") {
  Platform.Response.Redirect(redirectUrl, true);
}
</script>

<script>
  /* Browser-side JavaScript — handles instant screen switching. */

  function showScreen(name, stepperStep) {
    var screens = document.getElementsByClassName('screen');
    for (var i = 0; i < screens.length; i++) {
      screens[i].className = screens[i].className.replace(' active', '');
    }
    var target = document.getElementById('screen-' + name);
    if (target) { target.className += ' active'; }

    var stepper = document.getElementById('stepper');
    stepper.setAttribute('data-active', stepperStep);
    stepper.style.display = (name === 'done') ? 'none' : 'flex';
  }

  function chooseMode(mode) {
    document.getElementById('ddlMode').value = mode;
    showScreen(mode, '3');
  }

  function backToChoose() {
    document.getElementById('ddlMode').value = '';
    showScreen('choose', '2');
  }

  function startOver() {
    window.location.href = window.location.href.split('?')[0];
  }

  /* Decide which screen to show, based on what the server logic worked out. */
  var serverStep = "%%=v(@renderStep)=%%";

  if (serverStep === "choose") {
    showScreen('choose', '2');
  } else if (serverStep === "add") {
    document.getElementById('ddlMode').value = 'add';
    showScreen('add', '3');
  } else if (serverStep === "update") {
    document.getElementById('ddlMode').value = 'update';
    showScreen('update', '3');
  } else if (serverStep === "done") {
    showScreen('done', '3');
  } else {
    showScreen('email', '1');
  }
</script>
