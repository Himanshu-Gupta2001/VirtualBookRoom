SELECT
    jm.JourneyID,
    jm.JourneyName,
    CAST(jm.Version AS VARCHAR(50)) AS VersionID,
    em.LastSendDate,
    jm.ModifiedDate AS LastModifiedDate
FROM [Journey_Master] jm
INNER JOIN (
    SELECT
        cja.JourneyID,
        MAX(s.EventDate) AS LastSendDate
    FROM [Journey_Activity] cja
    LEFT JOIN ENT._JourneyActivity dja
        ON dja.ActivityID = cja.ActivityID
    LEFT JOIN ENT._Sent s
        ON s.TriggererSendDefinitionObjectID = dja.JourneyActivityObjectID
    WHERE cja.Type IN ('EMAILV2', 'EmailAudience')
    GROUP BY cja.JourneyID
) em
    ON em.JourneyID = jm.JourneyID
WHERE
    jm.ModifiedDate < DATEADD(DAY, -180, GETDATE())
    AND (
        em.LastSendDate IS NULL
        OR em.LastSendDate < DATEADD(DAY, -180, GETDATE())
    )
