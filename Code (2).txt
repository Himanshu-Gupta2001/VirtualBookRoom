<script runat="server">
Platform.Load("core", "1.1.1");

// -------------------------------------------------------------------------
// CONFIGURATION
// -------------------------------------------------------------------------
var myDE = DataExtension.Init("9827D2D9-E329-405F-9EC1-51855AD68D44"); 

try {
    var deRows = myDE.Rows.Retrieve();

    Write("<h2>Execution Summary: " + (deRows ? deRows.length : 0) + " rows retrieved</h2>");
    Write("<table border='1' cellpadding='6' cellspacing='0' style='border-collapse:collapse; font-family:sans-serif;'>");
    Write("<tr style='background-color:#f2f2f2;'><th>Row #</th><th>ContactKey</th><th>Trimmed Key</th><th>SF Row Count</th><th>Updated Status</th></tr>");

    if (deRows && deRows.length > 0) {
        for (var i = 0; i < 5; i++) {
            var row = deRows[i];
            var rawContactKey = row.ContactKey;

            if (!rawContactKey) {
                Write("<tr><td>" + (i + 1) + "</td><td colspan='4' style='color:orange;'>Skipped: Empty ContactKey</td></tr>");
                continue;
            }

            // Clean leading/trailing spaces
            var cleanContactKey = String(rawContactKey).replace(/^\s+|\s+$/g, "");

            // -----------------------------------------------------------------
            // 1. Safe AMPscript - Only fetch the row count
            // -----------------------------------------------------------------
            var amp = ""
                + "%" + "%[\\n"
                + "VAR @contactKey, @sfContactRows, @rowCount\\n\\n"
                + "SET @contactKey = '" + cleanContactKey + "'\\n\\n"
                
                + "/* Search the Contact object */\\n"
                + "SET @sfContactRows = RetrieveSalesforceObjects('Contact', 'Id', 'Id', '=', '0034W00002oCwUEQA0')\\n"
                + "SET @rowCount = RowCount(@sfContactRows)\\n"
                + "]%" + "%\\n";

            // -----------------------------------------------------------------
            // 2. Execute AMPscript safely
            // -----------------------------------------------------------------
            Platform.Function.TreatAsContent(amp);
            
            var rowCountStr = Variable.GetValue("@rowCount");
            var rowCount = parseInt(rowCountStr, 10) || 0;
            var isFoundBool = (rowCount > 0);

            // -----------------------------------------------------------------
            // 3. Update the Data Extension
            // -----------------------------------------------------------------
            myDE.Rows.Update(
                { "Found": isFoundBool },
                ["ContactKey"],
                [rawContactKey]
            );

            // -----------------------------------------------------------------
            // 4. Print row-level result to the screen
            // -----------------------------------------------------------------
            var statusColor = isFoundBool ? "green" : "red";
            Write("<tr>");
            Write("<td>" + (i + 1) + "</td>");
            Write("<td>" + rawContactKey + "</td>");
            Write("<td>" + cleanContactKey + "</td>");
            Write("<td>" + rowCount + "</td>");
            Write("<td style='color:" + statusColor + "; font-weight:bold;'>" + isFoundBool + "</td>");
            Write("</tr>");
        }
    }

    Write("</table>");

} catch (err) {
    Write("<div style='color:red; font-family:sans-serif;'>");
    Write("<h3>Execution Error:</h3>");
    Write("<p><strong>Message:</strong> " + err.message + "</p>");
    if (err.description) {
        Write("<p><strong>Description:</strong> " + err.description + "</p>");
    }
    Write("</div>");
}
</script>
