          RequestParameter() and again with SSJS Request.GetFormField().
          The request stream can only be consumed once; the second read threw
          the same error. That is why the page rendered fine on a plain load

<script runat="server">
Platform.Load("Core", "1.1.1");

/* ============================================================================
   SERVER-SIDE BUSINESS LOGIC

   RULES THIS BLOCK OBEYS — do not break them when editing:
   1. NO Request.* calls. The request was already read in AMPscript above.
      Reading it a second time throws the security descriptor error.
   2. NO Platform.Function.Trim / GUID / Now / Stringify. Trim is broken in
      this BU; the others are routed around as a precaution.
   3. NO nested try/catch, and no function declared inside a block.
   ============================================================================ */

/* true = print the real error on the page. MUST be false before go-live. */
var DEBUG = true;

/* ============================ TEST ON THE PUBLISHED URL =====================
   Content Builder's PREVIEW pane runs pages in a restricted "content" context
   where InsertData / UpdateData / Lookup are blocked outright. Preview will
   always report:
     "InsertData Function is not valid in content. This function is only
      allowed in a non batch context."  (OMM_FUNC_CONTEXT_ERR)
   That is not a bug in this page. Publish the page and test on its real URL.
   ========================================================================== */

/* The "ENT." prefix is ONLY correct if these are shared / enterprise DEs.
   If they live in the local Business Unit, remove the prefix. */
var deDetails = "ENT.API_Details_GEE-61699";
var deAudit   = "ENT.API_Credential_Audit_Log";

/* Set false once Client Id / Client Secret / Authentication Base URI are
   marked nullable in the details DE. */
var deHasRequiredPlainColumns = true;

var dbgStep = "start";

/* ============================ HELPER FUNCTIONS ============================ */

/* Plain-JScript trim.
   Platform.Function.Trim() throws the security descriptor error in this BU,
   AND this engine (Jint) throws "StartIndex cannot be less than zero" on
   String.substring(). So this builds the result one character at a time —
   no substring, no slice, no regex. */
function isWs(c) {
  return (c == " " || c == "\t" || c == "\r" || c == "\n");
}

function trimStr(v) {
  if (v == null) { return ""; }
  var s = String(v);
  if (s.length == 0) { return ""; }
  var out = "";
  var pending = "";
  var started = false;
  for (var i = 0; i < s.length; i++) {
    var c = s.charAt(i);
    if (isWs(c)) {
      if (started) { pending = pending + c; }
    } else {
      if (pending != "") { out = out + pending; pending = ""; }
      out = out + c;
      started = true;
    }
  }
  return out;
}

/* Truncate without substring, for the 4000-char ErrorMessage column. */
function cut(v, maxLen) {
  if (v == null) { return ""; }
  var s = String(v);
  if (s.length <= maxLen) { return s; }
  var out = "";
  for (var i = 0; i < maxLen; i++) { out = out + s.charAt(i); }
  return out;
}

/* Read an AMPscript variable as a trimmed string. This replaces every
   Request.GetFormField / GetQueryStringParameter call the page used to make. */
function av(name) {
  var v = Variable.GetValue(name);
  if (v == null) { return ""; }
  return trimStr(v);
}

/* Read an AMPscript variable that holds a number. */
function avNum(name) {
  var v = Variable.GetValue(name);
  if (v == null || v == "") { return 0; }
  var n = parseInt(String(v), 10);
  if (isNaN(n)) { return 0; }
  return n;
}

function len(v) {
  if (v == null) { return 0; }
  return String(v).length;
}

/* Escape values before printing them back into HTML. split/join, not regex. */
function esc(v) {
  if (v == null) { return ""; }
  var s = String(v);
  s = s.split("&").join("&amp;");
  s = s.split("<").join("&lt;");
  s = s.split(">").join("&gt;");
  s = s.split('"').join("&quot;");
  return s;
}

/* Describe an exception without Platform.Function.Stringify. */
function errText(ex) {
  if (ex == null) { return "unknown error"; }
  var m = "";
  if (ex.message != null)     { m = m + ex.message; }
  if (ex.description != null) { m = m + " | " + ex.description; }
  if (m == "")                { m = String(ex); }
  return m;
}

/* Audit entries are queued in memory here and written to the DE later, after
   the main try/catch has closed — so no try is ever nested inside another. */
var auditQueue = [];

function queueAudit(actionType, pkgName, bu, who, status, errMsg) {
  auditQueue[auditQueue.length] = {
    a: actionType, p: pkgName, b: bu, w: who, s: status, e: errMsg
  };
}

/* Insert a new package. Only encrypted values are stored. */
function insertPackage(packageName, encId, encSecret, encAuth, encRest, bu, scope, expDt, nowVal) {
  var cols = ["API Name",
              "Encrypted Client Id","Encrypted Client Secret",
              "Encrypted Authentication Base URI","Encrypted REST Base URI",
              "Business Unit","Scope",
              "Client Secret Expiration Date","Last Modified Date"];

  var vals = [packageName,
              encId, encSecret,
              encAuth, encRest,
              bu, scope,
              expDt, nowVal];

  /* Placeholders purely to satisfy "Nullable = No" on the plain-text columns. */
  if (deHasRequiredPlainColumns) {
    cols.push("Client Id");
    cols.push("Client Secret");
    cols.push("Authentication Base URI");
    vals.push("[ENCRYPTED]");
    vals.push("[ENCRYPTED]");
    vals.push("[ENCRYPTED]");
  }

  Platform.Function.InsertData(deDetails, cols, vals);
}

/* Update only the rotated secret, its expiry, and the last-modified stamp. */
function updatePackageSecret(packageName, encSecret, newExpiration, nowVal) {
  Platform.Function.UpdateData(
    deDetails,
    ["API Name"], [packageName],
    ["Encrypted Client Secret", "Client Secret Expiration Date", "Last Modified Date"],
    [encSecret, newExpiration, nowVal]
  );
}

/* ============================ PAGE STATE ============================ */

var successMsg  = "";
var errorMsg    = "";
var renderStep  = "email";   // email | choose | add | update | done
var redirectUrl = "";

var ampGuid = "", ampNow = "";
var pageUrl = "", emailAddress = "";
var isValidUser = false;
var isVerify = "", isSubmit = "", mode = "";

var encClientId = "", encClientSecret = "", encAuthBaseURI = "", encRestBaseURI = "", encNewSecret = "";

var f_packageName = "", f_authBaseURI = "", f_restBaseURI = "", f_expirationDate = "";
var f_businessUnit = "", f_scope = "", f_packageNameUpdate = "", f_newExpiration = "";

var clientIdLen = 0, clientSecretLen = 0, newSecretLen = 0;
var qsStatus = "", qsAction = "", qsPkg = "";
var packageName = "", packageName2 = "", businessUnit2 = "";
var existing = null, rows = null;

/* ==================== MAIN LOGIC — THE ONLY try/catch ==================== */

try {

  dbgStep = "read-ampscript-vars";

  ampGuid      = av("@ampGuid");
  ampNow       = Variable.GetValue("@ampNow");
  pageUrl      = av("@PageURL");
  emailAddress = av("@EmailAddress");
  isValidUser  = (av("@isValidUser") == "true");

  isVerify = av("@isVerifyAmp");
  isSubmit = av("@isSubmitAmp");
  mode     = av("@Mode");

  encClientId     = Variable.GetValue("@EncClientID");
  encClientSecret = Variable.GetValue("@EncClientSecret");
  encAuthBaseURI  = Variable.GetValue("@EncAuthBaseURI");
  encRestBaseURI  = Variable.GetValue("@EncRestBaseURI");
  encNewSecret    = Variable.GetValue("@EncNewSecret");

  dbgStep = "read-form-values";

  f_packageName       = av("@f_packageName");
  f_authBaseURI       = av("@f_authBaseURI");
  f_restBaseURI       = av("@f_restBaseURI");
  f_expirationDate    = av("@f_expirationDate");
  f_businessUnit      = av("@f_businessUnit");
  f_scope             = av("@f_scope");
  f_packageNameUpdate = av("@f_packageNameUpdate");
  f_newExpiration     = av("@f_newExpiration");

  clientIdLen     = avNum("@p_clientIdLen");
  clientSecretLen = avNum("@p_clientSecretLen");
  newSecretLen    = avNum("@p_newSecretLen");

  dbgStep = "read-query-string";

  qsStatus = av("@p_status");
  qsAction = av("@p_action");
  qsPkg    = av("@p_pkg");

  if (qsStatus == "success") {
    renderStep = "done";
    if (qsAction == "add") {
      successMsg = "Package '" + qsPkg + "' was created successfully. All sensitive values were AES-encrypted before storage.";
    } else if (qsAction == "update") {
      successMsg = "Client secret for '" + qsPkg + "' was updated successfully.";
    }
  }

  // ---- SCENARIO 1: "Verify & Continue" -----------------------------------
  if (isVerify == "Verify") {

    dbgStep = "verify-branch";

    if (len(emailAddress) == 0) {
      errorMsg = "Please enter your email address.";
      renderStep = "email";
    } else if (!isValidUser) {
      errorMsg = "We couldn't verify '" + emailAddress + "' as an active Salesforce user. Please check the address and try again.";
      queueAudit("Access Denied", "", "", emailAddress, "Failed", "Email not found or inactive in Salesforce User object");
      renderStep = "email";
    } else {
      renderStep = "choose";
    }

  // ---- SCENARIO 2: "Save Securely" ---------------------------------------
  } else if (isSubmit == "Submit") {

    dbgStep = "submit-branch";

    if (len(emailAddress) == 0 || !isValidUser) {

      errorMsg = "Your verification could not be confirmed. Please re-enter your email address and verify again.";
      renderStep = "email";
      if (len(emailAddress) > 0) {
        queueAudit("Access Denied", "", "", emailAddress, "Failed", "Re-verification failed on final submit");
      }

    // ---- PATH A: Add a new package ---------------------------------------
    } else if (mode == "add") {

      dbgStep = "add-path";
      renderStep = "add";
      packageName = f_packageName;

      /* Secret lengths come from AMPscript — the raw secrets never enter SSJS. */
      if (len(packageName) == 0 || clientIdLen == 0 || clientSecretLen == 0 ||
          len(f_authBaseURI) == 0 || len(f_restBaseURI) == 0 || len(f_expirationDate) == 0 ||
          len(f_businessUnit) == 0 || len(f_scope) == 0) {

        errorMsg = "All fields are required to add a new package.";

      } else {

        dbgStep = "add-duplicate-check";
        existing = Platform.Function.Lookup(deDetails, "API Name", "API Name", packageName);

        if (existing != null && String(existing).length > 0) {

          errorMsg = "A package named '" + packageName + "' already exists. Please use Update instead.";
          queueAudit("Create", packageName, f_businessUnit, emailAddress, "Failed", "Duplicate package name");

        } else {

          dbgStep = "add-insert";
          insertPackage(packageName, encClientId, encClientSecret, encAuthBaseURI,
                        encRestBaseURI, f_businessUnit, f_scope, f_expirationDate, ampNow);

          queueAudit("Create", packageName, f_businessUnit, emailAddress, "Success", "");

          /* Remember the URL only. Redirect() terminates the request by
             throwing internally, so it must happen clear of any try/catch. */
          redirectUrl = pageUrl + "?status=success&action=add&pkg=" + encodeURIComponent(packageName);
        }
      }

    // ---- PATH B: Update an existing package ------------------------------
    } else if (mode == "update") {

      dbgStep = "update-path";
      renderStep = "update";
      packageName2 = f_packageNameUpdate;

      if (len(packageName2) == 0 || newSecretLen == 0 || len(f_newExpiration) == 0) {

        errorMsg = "Package Name, New Client Secret, and New Expiration Date are all required.";

      } else {

        dbgStep = "update-lookup";
        rows = Platform.Function.LookupRows(deDetails, "API Name", packageName2);

        if (rows == null || rows.length == 0) {

          errorMsg = "No package found named '" + packageName2 + "'. Please check the Package Name and try again.";
          queueAudit("Update", packageName2, "", emailAddress, "Failed", "Package not found");

        } else {

          businessUnit2 = rows[0]["Business Unit"];
          if (businessUnit2 == null) { businessUnit2 = ""; }

          dbgStep = "update-write";
          updatePackageSecret(packageName2, encNewSecret, f_newExpiration, ampNow);

          queueAudit("Update", packageName2, businessUnit2, emailAddress, "Success", "");

          redirectUrl = pageUrl + "?status=success&action=update&pkg=" + encodeURIComponent(packageName2);
        }
      }

    } else {
      renderStep = "choose";
    }
  }

} catch (pageEx) {

  redirectUrl = "";   // never redirect off the back of a failure

  if (dbgStep == "add-insert" || dbgStep == "add-duplicate-check") {
    renderStep = "add";
    errorMsg = "An unexpected error occurred while creating the package. Please contact your Marketing Cloud administrator.";
  } else if (dbgStep == "update-lookup" || dbgStep == "update-write") {
    renderStep = "update";
    errorMsg = "An unexpected error occurred while updating the package. Please contact your Marketing Cloud administrator.";
  } else {
    renderStep = "email";
    errorMsg = "A technical error occurred. Please contact your Marketing Cloud administrator.";
  }

  queueAudit("Error", "", "", emailAddress, "Failed", cut("step=" + dbgStep + " | " + errText(pageEx), 3800));

  if (DEBUG) {
    Write("<pre style='background:#FDECEC;color:#9A2626;padding:12px;font-size:12px;white-space:pre-wrap;'>"
          + "FAILED AT STEP: " + dbgStep + "\n" + errText(pageEx) + "</pre>");
  }
}

/* ---- WRITE THE QUEUED AUDIT ROWS ---------------------------------------
   Runs AFTER the try/catch above has closed, so its try is never nested.
   A logging failure here never affects what the user sees. */
try {
  if (ampGuid == "") { ampGuid = Variable.GetValue("@ampGuid"); }
  if (ampNow  == "") { ampNow  = Variable.GetValue("@ampNow"); }
  for (var qi = 0; qi < auditQueue.length; qi++) {
    Platform.Function.InsertData(
      deAudit,
      ["AuditID","ActionType","PackageName","BusinessUnit","PerformedBy","ActionDate","Status","ErrorMessage"],
      [ampGuid + "-" + qi, auditQueue[qi].a, auditQueue[qi].p, auditQueue[qi].b,
       auditQueue[qi].w, ampNow, auditQueue[qi].s, auditQueue[qi].e]
    );
  }
} catch (flushEx) {
  if (DEBUG) {
    Write("<pre style='background:#FFF6E5;color:#8A5A00;padding:12px;font-size:12px;white-space:pre-wrap;'>"
          + "AUDIT LOG WRITE FAILED (page logic itself was fine)\n" + errText(flushEx) + "</pre>");
  }
}

/* ---- HAND OFF TO THE HTML BELOW --------------------------------------- */
Variable.SetValue("@renderStep",   renderStep);
Variable.SetValue("@successMsg",   esc(successMsg));
Variable.SetValue("@errorMsg",     esc(errorMsg));
Variable.SetValue("@Mode",         esc(mode));
Variable.SetValue("@EmailAddress", esc(emailAddress));
Variable.SetValue("@f_packageName",       esc(f_packageName));
Variable.SetValue("@f_authBaseURI",       esc(f_authBaseURI));
Variable.SetValue("@f_restBaseURI",       esc(f_restBaseURI));
Variable.SetValue("@f_expirationDate",    esc(f_expirationDate));
Variable.SetValue("@f_businessUnit",      esc(f_businessUnit));
Variable.SetValue("@f_scope",             esc(f_scope));
Variable.SetValue("@f_packageNameUpdate", esc(f_packageNameUpdate));
Variable.SetValue("@f_newExpiration",     esc(f_newExpiration));

/* Now — and only now, clear of every try/catch — perform the redirect. */
if (redirectUrl != "") {
  Platform.Response.Redirect(redirectUrl, true);
}
</script>

<script>
  /* Browser-side JavaScript — handles instant screen switching. */

  function showScreen(name, stepperStep) {
    var screens = document.getElementsByClassName('screen');
    for (var i = 0; i < screens.length; i++) {
      screens[i].className = screens[i].className.replace(' active', '');
    }
    var target = document.getElementById('screen-' + name);
    if (target) { target.className += ' active'; }

    var stepper = document.getElementById('stepper');
    stepper.setAttribute('data-active', stepperStep);
    stepper.style.display = (name === 'done') ? 'none' : 'flex';
  }

  function chooseMode(mode) {
    document.getElementById('ddlMode').value = mode;
    showScreen(mode, '3');
  }

  function backToChoose() {
    document.getElementById('ddlMode').value = '';
    showScreen('choose', '2');
  }

  function startOver() {
    window.location.href = window.location.href.split('?')[0];
  }

  /* Decide which screen to show, based on what the server logic worked out. */
  var serverStep = "%%=v(@renderStep)=%%";

  if (serverStep === "choose") {
    showScreen('choose', '2');
  } else if (serverStep === "add") {
    document.getElementById('ddlMode').value = 'add';
    showScreen('add', '3');
  } else if (serverStep === "update") {
    document.getElementById('ddlMode').value = 'update';
    showScreen('update', '3');
  } else if (serverStep === "done") {
    showScreen('done', '3');
  } else {
    showScreen('email', '1');
  }
</script>
