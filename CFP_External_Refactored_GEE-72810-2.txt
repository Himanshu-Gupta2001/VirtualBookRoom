     CENTRALIZED FORM PROCESSOR (CFP) - EXTERNAL CLOUDPAGE
     Story: GEE-72810   (clean-up per spike GEE-35446 + fix for incident INC4979312)
     ==========================================================================================

     WHAT THIS PAGE DOES
       External websites / partners (for example Anteriad) POST a form submission to this CloudPage.
       The page validates it, finds or creates the person in Salesforce (Lead / Contact), updates
       consent, preferences and subscriber status in Marketing Cloud, triggers the Journey Builder
       journeys and finally sends the visitor to the confirmation page.

     HOW THE FILE IS BUILT (read this first if you are new)
       * The page mixes two languages: AMPscript (code between the percent-percent-open-square-bracket
         markers) and Server-Side JavaScript, SSJS (code inside script tags with runat="server").
       * Several stages use a try / catch that is split over blocks:
             a script block that ends with:      try {
             an AMPscript block that does the work
             a script block that starts with:    } catch (e) { CFP_Fail("STAGE_NAME", e); }
         DO NOT move, delete or re-indent the "try {" and "} catch" lines independently of each other.
         If anything inside the AMPscript fails, the catch writes the error log and returns the error.
       * Comments are written INSIDE the blocks on purpose. Never use HTML comments in this file:
         everything outside the blocks is sent to the caller, so an HTML comment would be visible to
         anyone calling the page (for example from Postman).
       * NEVER type the closing script tag or the AMPscript markers inside a comment: the platform would
         read them as real code and break the page.

     STAGES (in execution order)
        1  Read the posted parameters, validate them, write the staging record
        2  Salesforce: find the Lead/Contact via ContactPointEmail (CPE), or create a Lead
        3  Promote the staging record to Centralized_Form_Processor_DE
        4  Update the consent status in Salesforce (+ remove the email from the suppression list)
        5  Re-subscribe the person's other (associated) email addresses when consent was updated
        6  Read the person's name from Salesforce and save the preferences (Preferences 1-10)
        7  Make the subscriber Active in Marketing Cloud and decrypt the API credentials
        8  Trigger the journeys (dynamic journeys + the Centralized Form Processor journey)
        9  Safety net: make sure the Lead's consent status is "Opt In" (GEE-18206)
       10  Send the browser to the confirmation page

     GLOSSARY
        CPE        = Salesforce ContactPointEmail object (one row per email address of a person).
                     Field KPMG_Lead_Contact_Id__c holds the Id of the Lead or Contact owning the email.
        SFID       = Salesforce Id of the Lead (starts with 00Q) or Contact (starts with 003).
        IFD        = Salesforce object KPMG_Individual_Firm_Details__c (holds the consent status).

                     failed submission is never lost.

        Sync DE    = the Marketing Cloud synchronized Data Extension that mirrors Salesforce ContactPointEmail.
        PrimaryKey = reference such as EX-1A2B3C4D created for each submission. It is stored in the staging
                     record, the final record and every error-log row. Use it to follow one submission.

        * Inquiry Type handling, Contact Us journey, Internal Notification journey: these belong to the

        * The three copies of the Lead create / update logic: now ONE flow (stage 2).
        * Repeated ContactPointEmail lookups: now ONE lookup whose result is reused (stage 2).

     ========================================================================================== */

  /* Read an AMPscript variable (for example "FirstName" reads @FirstName) as text. Never returns null. */
  function CFP_Get(name) {
    var v = Variable.GetValue("@" + name);
    if (v === null || v === undefined) { return ""; }
    return String(v);
  }

  /* Cut a text to at most n characters (so it always fits the Data Extension column) */
  function CFP_Trunc(s, n) {
    s = String(s);
    return (s.length > n) ? s.substring(0, n) : s;
  }

  /* Turn any error (text or object) into readable text for the error log */
  function CFP_ErrText(e) {
    if (e === null || e === undefined) { return "Unknown error"; }
    if (typeof e === "string") { return e; }
    var t = "";
    try { if (e.message) { t += String(e.message); } } catch (x1) {}
    try { if (e.description) { t += (t ? " | " : "") + String(e.description); } } catch (x2) {}
    try { var s = Stringify(e); if (s && s !== "{}") { t += (t ? " | " : "") + s; } } catch (x3) {}
    return t ? t : String(e);
  }

  /* The wait routine of the old page (GSENHANCEM-15677) - intentionally unchanged.
     SSJS has no sleep command, so it loops until the time has passed. */
  function CFP_Sleep(milliseconds) {
    var start = new Date().getTime();
    for (var i = 0; i < 1e7; i++) {
      if ((new Date().getTime() - start) > milliseconds) { break; }
    }
  }

  /* Check every posted field against CFP_FIELDS. Returns a list of problems (empty list = all valid).
     Only field NAMES are returned, never the values. */
  function CFP_ValidateInput() {
    var bad = [];
    for (var i = 0; i < CFP_FIELDS.length; i++) {
      var name = CFP_FIELDS[i][0], max = CFP_FIELDS[i][1], type = CFP_FIELDS[i][2];
      var v = CFP_Get(name);
      if (v.length > max) { bad.push(name + " exceeds " + max + " characters"); }
      if (type === 1 && CFP_CHARSET_STD.test(v)) { bad.push(name + " has disallowed characters"); }
      if (type === 2 && CFP_CHARSET_MSG.test(v)) { bad.push(name + " has disallowed characters"); }
      if (type === 3 && v.indexOf("alert(") > -1) { bad.push(name + " contains a script pattern"); }
    }
    return bad;
  }

  /* ERROR LOG WRITER (Scenario 6) - the ONE place that writes Centralized_Form_Processor_DE_Error_Log.
     It stores every posted value that is known at the moment of the failure (all fields of CFP_FIELDS),
     plus RFP_Attachments, PrimaryKey, CFP_PK (GUID that links to the staging record), Page_Type = External
     and ErrorMessage. ErrorMessage looks like:
         [STAGE] description | HTTP=<status> | SFID=<Salesforce Id if known> | LOOKUP=<how the Id was found> | t=+<ms>
     level:  "ERROR"          = the submission failed and the caller gets an error
             "ERROR_NONFATAL" = something failed but the submission continues (as it always did)
     This function never throws, so a logging problem can never hide the real error. If the full row
     cannot be written, a smaller row (name, email, message) is written instead. */
  function CFP_LogError(stage, detail, level, httpStatus) {
    var status = httpStatus ? String(httpStatus) : "";
    var msg = "[" + stage + "] " + CFP_ErrText(detail) +
              " | HTTP=" + status +
              " | SFID=" + CFP_Get("Salesforce_ID") +
              " | LOOKUP=" + CFP_Get("Resolved_From") +
              " | t=+" + (new Date().getTime() - CFP_START) + "ms" +
              ((level === "ERROR_NONFATAL") ? " | NON-FATAL (submission continued)" : "");
    msg = CFP_Trunc(msg, 3999);
    var names = [], values = [];
    for (var i = 0; i < CFP_FIELDS.length; i++) {
      names.push(CFP_FIELDS[i][0]);
      values.push(CFP_Trunc(CFP_Get(CFP_FIELDS[i][0]), CFP_FIELDS[i][1]));
    }
    names.push("RFP_Attachments"); values.push(CFP_Trunc(CFP_Get("RFP_Attachments"), 50));
    names.push("PrimaryKey");      values.push(CFP_Trunc(CFP_Get("PrimaryKey"), 50));
    names.push("CFP_PK");          values.push(CFP_Trunc(CFP_Get("guid"), 50));
    names.push("Page_Type");       values.push("External");
    names.push("ErrorMessage");    values.push(msg);
    try {
      Platform.Function.InsertData(CFP_ERROR_LOG_DE, names, values);
    } catch (eLog) {
      try {
        Platform.Function.InsertData(CFP_ERROR_LOG_DE,
          ["FirstName","LastName","EmailAddress","ErrorMessage"],
          [CFP_Trunc(CFP_Get("FirstName"),100), CFP_Trunc(CFP_Get("LastName"),100),
           CFP_Trunc(CFP_Get("EmailAddress"),254), msg]);
      } catch (eLog2) { /* nothing more can be done - never block the response */ }
    }
  }

  /* Give the caller the error response, according to CFP_ERROR_MODE (see the configuration above).
     The text always contains the PrimaryKey so support can find the matching error-log row. */
  function CFP_ErrorResponse(httpStatus, stage) {
    var ref = "CFP error " + httpStatus + " [" + stage + "] ref " + CFP_Get("PrimaryKey");
    if (CFP_ERROR_MODE === "REDIRECT") {
      Platform.Response.Redirect(CFP_ERROR_URL);
    } else if (CFP_ERROR_MODE === "RAISE_ERROR") {
      Platform.Function.RaiseError(ref, true, "statusCode", String(httpStatus));
    } else {
      throw ref;    /* mode THROW: unhandled script error -> HTTP 500 error response */
    }
  }

  /* Standard failure handling used by every stage's catch block:
     1) write the error log row, 2) return the error to the caller. */
  function CFP_Fail(stage, e) {
    var st = CFP_STATUS_MAP[stage] || 500;
    CFP_LogError(stage, e, "ERROR", st);
    CFP_ErrorResponse(st, stage);
  }
</script>

     STAGE 1b - VALIDATE THE SUBMISSION
     Same rules as the previous version of the page. If anything is invalid, nothing is saved:
     the problem is written to the error log (field names only) and the caller gets the error.
     The decision is taken first and the error is returned AFTER the try / catch, so that the error

  var cfpRejectMsg = "";
  try {
    var cfpFailures = CFP_ValidateInput();

    }
  } catch (eVal) {
    cfpRejectStage = "INPUT_VALIDATION";
    cfpRejectMsg = "Validation routine failed: " + CFP_ErrText(eVal);
  }
  if (cfpRejectStage !== "") {
    CFP_LogError(cfpRejectStage, cfpRejectMsg, "ERROR", 400);
    CFP_ErrorResponse(400, cfpRejectStage);
  }

<script runat="server">
  /* ========================================================================================
     STAGE 8 - JOURNEY BUILDER
     Same approach as the Internal CFP backend page:
       a) authenticate with the decrypted API credentials and get an access token,
       b) trigger the DYNAMIC journeys of this form (event ids from FormName_Event_ID_Mapping_DE, GEE-9867),
       c) read the event key of the Centralized Form Processor journey from
          ENT.Journey_API_Event_Key_Config_DE (row of this business unit's MID, column
          CentralizedFormProcessor_Journey) and trigger that journey.
     The Contact Us journey and the Internal Notification journey do NOT exist on the External page.
     Every journey receives the same data: all form fields + EmailAddress, PK_Email, Salesforce_ID.
     Failure rules: a failing DYNAMIC journey is logged and the page continues (as before); a failure of
     the token / configuration lookup / Centralized Form journey call fails the submission.
     ======================================================================================== */
  var jContentType = "application/json";

  /* The data sent with every journey entry (all form fields, same list as the Internal CFP page except
     Inquiry_Type, which does not exist on the External page). */
  function CFP_BuildJourneyData() {
    var cols = ["FirstName","LastName","Company","JobTitle","Country","Phone","CID","medium","MID","term",
                "Campaign","Source","content","db_annual_sales","db_city","db_company_name","db_country_name",
                "db_employee_count","db_industry","db_primary_naics","db_primary_sic","db_state",
                "db_street_address","db_sub_industry","db_web_site","db_zip","confirmationPage","formName",
                "pageURL","Full_Page_URL","conditionalProcessing","Preference1","Preference2","Preference3",
                "Preference4","Preference5","Preference6","Preference7","Preference8","Preference9",
                "Preference10","Tactic_ID","Department","Level","Function","Message","Lead_Source",
                "Lead_Source_Detail","RFP_Attachments","marketingTheme","marketingCampaign",
                "marketingSubCampaign","Salesforce_ID"];
    var data = { "EmailAddress": CFP_Get("EmailAddress"), "PK_Email": CFP_Get("PK_Email") };
    for (var i = 0; i < cols.length; i++) { data[cols[i]] = CFP_Get(cols[i]); }
    return data;
  }

  /* Put the person into one journey. A response other than 201 (Created) is written to the error log as
     NON-FATAL (the old page ignored it silently). HTTP.Post errors are thrown to the caller. */
  function CFP_FireJourney(eventsUrl, eventKey, headerNames, headerValues, stage) {
    var body = { "ContactKey": CFP_Get("EmailAddress"), "EventDefinitionKey": eventKey, "Data": CFP_BuildJourneyData() };
    var res = HTTP.Post(eventsUrl, jContentType, Stringify(body), headerNames, headerValues);
    if (res.StatusCode != 201) {
      var resText = "";
      try { resText = CFP_Trunc(String(res.Response[0]), 500); } catch (eRes) {}
      CFP_LogError(stage, "Journey entry returned HTTP " + res.StatusCode + " for event " + eventKey + " : " + resText, "ERROR_NONFATAL", res.StatusCode);
    }
  }

  try {
    var jMID = CFP_Get("MID_Value");   /* Marketing Cloud business unit id, set in stage 7 */

    /* (a) access token - credentials were decrypted in stage 7 */
    var jAccessToken = null;
    var jRestUrl = null;
    var jPayload = {
      client_id: CFP_Get("Client_ID_DEC"),
      client_secret: CFP_Get("Client_Secret_DEC"),
      grant_type: "client_credentials",
      account_id: jMID
    };
    var jTokenReq = HTTP.Post(CFP_Get("Authentication_Base_URI_DEC") + "/v2/token", jContentType, Stringify(jPayload));
    if (jTokenReq.StatusCode == 200) {
      var jTokenResp = Platform.Function.ParseJSON(jTokenReq.Response[0]);
      jAccessToken = jTokenResp.access_token;
      jRestUrl = jTokenResp.rest_instance_url;
    } else {
      /* Journeys cannot be triggered without a token. As before the page continues, but the problem is logged. */
      CFP_LogError("JOURNEY_AUTH", "Access token request returned HTTP " + jTokenReq.StatusCode + " - journeys were not triggered", "ERROR_NONFATAL", jTokenReq.StatusCode);
    }

    if (CFP_Get("EmailAddress") !== "" && jAccessToken !== null && jAccessToken !== undefined) {
      var jHeaderNames = ["Authorization"];
      var jHeaderValues = ["Bearer " + jAccessToken];
      var jEventsUrl = String(jRestUrl);
      if (jEventsUrl.charAt(jEventsUrl.length - 1) !== "/") { jEventsUrl += "/"; }
      jEventsUrl += "interaction/v1/events";

      /* (b) DYNAMIC JOURNEYS: FormName_Event_ID_Mapping_DE maps a formName to one or more journey event ids.
         One failing journey must not stop the others, so each one has its own try / catch. */
      try {
        var jMapDE = DataExtension.Init(CFP_JOURNEY_MAP_DE);
        var jMapRows = jMapDE.Rows.Lookup(["formName"], [CFP_Get("formName")]);
        if (jMapRows === null || jMapRows === undefined) { jMapRows = []; }
        for (var j = 0; j < jMapRows.length; j++) {
          try {
            CFP_FireJourney(jEventsUrl, jMapRows[j].Event_ID, jHeaderNames, jHeaderValues, "JOURNEY_DYNAMIC");
          } catch (eDyn) {
            CFP_LogError("JOURNEY_DYNAMIC", eDyn, "ERROR_NONFATAL");
          }
        }
      } catch (eDynInit) {
        CFP_LogError("JOURNEY_DYNAMIC", eDynInit, "ERROR_NONFATAL");
      }

      /* (c) CENTRALIZED FORM PROCESSOR JOURNEY - event key comes from the configuration DE, never hard-coded */
      var jCfgRows = Platform.Function.LookupRows(CFP_JOURNEY_CONFIG_DE, "MID", jMID);
      var jCfpKey = "";
      if (jCfgRows && jCfgRows.length > 0) {
        jCfpKey = jCfgRows[0]["CentralizedFormProcessor_Journey"] || "";
      }
      if (jCfpKey !== "") {
        CFP_FireJourney(jEventsUrl, jCfpKey, jHeaderNames, jHeaderValues, "JOURNEY_CFP");
      } else {
        CFP_LogError("JOURNEY_CFP", "No CentralizedFormProcessor_Journey event key found in " + CFP_JOURNEY_CONFIG_DE + " for MID " + jMID + " - journey was not triggered", "ERROR_NONFATAL");
      }
    }
  } catch (eJourney) {
    CFP_Fail("JOURNEY_PROCESSING", eJourney);
  }
</script>
