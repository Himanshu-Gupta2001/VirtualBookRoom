     (a) SESSION KEY  - an AES-encrypted expiry timestamp we minted ourselves

                        or on the query string. Valid for 60 minutes.

     (b) AUTH CODE    - a one-time ?code= handed to us by the Marketing Cloud
                        authorization server. We swap it for a token to prove
                        it is real, then immediately trade it for a session
                        key so the code never sits in the address bar.

   Neither one? -> Redirect() to the authorization server, which shows the
   Marketing Cloud login screen and bounces the user straight back here.

   WHY THE SESSION KEY MATTERS ON THIS PARTICULAR PAGE
   ---------------------------------------------------

  SET @authTenant       = "mchhklk0drvdbjf-g54n6vj27bky"

  SET @authClientId     = "b1t7dahx01semsqssqzs2fa2"

     Whole HOURS only. v8 used DateAdd(Now(), 30, "MI") and that is the single
     change between "reached the form" (v7) and "redirect loop" (v8). If the
     "MI" interval misbehaves, DateAdd yields nothing, FormatDate yields an

     the page redirects forever. Not worth the risk for a finer dial.

     The real freshness guarantee is the /v2/authorize bounce, which checks the
     user's LIVE session cookie at that instant. This value only controls how
     long we avoid re-checking it. */
  SET @sessionHours = 1

  SET @encIV   = "00000000000000000000000000000000"

  /* ---- v7: TRANSPORT-SAFE KEY ---------------------------------------------
     EncryptSymmetric returns standard Base64, whose alphabet includes + / and =.
     CloudPages does not round-trip "+" reliably through a form POST or a query
     string - it very often arrives as a SPACE. The length stays 24 either way,
     so a length check cannot catch it, and DecryptSymmetric then chokes on the
     corrupted string. That is what silently invalidates the session on submit
     and bounces the user back to the login server.

     So the key never travels as raw Base64. On the way out we map:
         +  ->  -        /  ->  _        =  ->  ~
     and on the way in we map straight back. Those three replacement characters
     can never occur in Base64 output, so the mapping is lossless, and the
     result survives any amount of URL encoding or decoding untouched.
     Same string is used in the hidden field AND on the query string. */
  SET @passedSessionKey = @rawSessionKey

    SET @passedSessionKey = Replace(@passedSessionKey, "-", "+")
    SET @passedSessionKey = Replace(@passedSessionKey, "_", "/")
    SET @passedSessionKey = Replace(@passedSessionKey, "~", "=")

     The plaintext is a fixed-width 14-character stamp (yyyyMMddHHmmss). Fixed

          us reject a malformed key BEFORE handing it to DecryptSymmetric
          (DecryptSymmetric throws a hard 500 on garbage, and AMPscript has no
          try/catch to save you);
       2. two stamps in this format compare correctly with > and <=. */
  SET @newSessionKeyPlain = FormatDate(DateAdd(Now(), @sessionHours, "H"), "yyyyMMddHHmmss")

  SET @sessionKeyUrl = @sessionKey

  SET @sessionKeyUrl = Replace(@sessionKeyUrl, "/", "_")
  SET @sessionKeyUrl = Replace(@sessionKeyUrl, "=", "~")

  SET @nowStamp = FormatDate(Now(), "yyyyMMddHHmmss")

     Decrypt the key we just created, in this same request, and check we get
     back exactly what we put in. This cannot throw - the input is something we
     encrypted a line ago - and it isolates the fault instantly:

       FAILED -> FormatDate / DateAdd / EncryptSymmetric is the problem. Nothing
                 to do with redirects, the browser, or the auth server.
       OK     -> the crypto is sound, so any session failure is transport.

     When it fails the page says so on screen, because it cannot work anyway. */
  SET @selfTestDecrypted = DecryptSymmetric(@sessionKey, "AES", "", @encPwd, "", @encSalt, "", @encIV)
  SET @selfTestResult = "FAILED"
  IF @selfTestDecrypted == @newSessionKeyPlain AND @newSessionKeyPlain != "" THEN
    SET @selfTestResult = "OK"
  ENDIF

  SET @sessionValid = "false"

    /* In the future (not expired) AND not further out than 60 minutes. */
    IF @decryptedExpiry > @nowStamp AND @decryptedExpiry <= @maxStamp THEN
      SET @sessionValid = "true"
    ENDIF
  ENDIF

  SET @authOK    = @sessionValid
  SET @authFatal = "false"
  SET @authDebug = CONCAT("sessionValid=", @sessionValid,
                          " | keyIn=[", @rawSessionKey, "]",
                          " | keyLenIn=", Length(@passedSessionKey),
                          " | keyLenExpected=", @expectedKeyLen,
                          " | decrypted=[", @decryptedExpiry, "]",
                          " | nowStamp=", @nowStamp,
                          " | maxStamp=", @maxStamp,

                          " | selfTest=", @selfTestResult,
                          " | plainOut=[", @newSessionKeyPlain, "]",
                          " | selfTestBack=[", @selfTestDecrypted, "]")

<script runat="server">

     1. NO Request.* calls  - everything arrives via Variable.GetValue().

     3. NO nested try/catch - the two try blocks below are SEQUENTIAL siblings.
   ============================================================================ */

var g_authOK   = (Variable.GetValue("@authOK") == "true");
var g_code     = Variable.GetValue("@code");
var g_tenant   = Variable.GetValue("@authTenant");
var g_clientId = Variable.GetValue("@authClientId");
var g_secret   = Variable.GetValue("@authClientSecret");
var g_pageUrl  = Variable.GetValue("@authPageURL");
var g_mcEmail  = Variable.GetValue("@mcUserEmail");
var g_dbg      = Variable.GetValue("@authDebug");
var g_tokenRaw = "";

if (g_code    == null) { g_code    = ""; }
if (g_mcEmail == null) { g_mcEmail = ""; }
if (g_dbg     == null) { g_dbg     = ""; }

/* Strip anything that would break the hand-built JSON body below. */
function jsonSafe(v) {
  if (v == null) { return ""; }
  var s = String(v);
  var out = "";
  for (var i = 0; i < s.length; i++) {
    var c = s.charAt(i);
    if (c != '"' && c != "\\" && c != "\n" && c != "\r" && c != "\t") { out = out + c; }
  }
  return out;
}

/* ---- TRY 1 of 2: trade the code for a token -------------------------------
   A HTTP 200 from /v2/token is by itself proof the code was genuine, so the
   security decision never depends on parsing the response body. That keeps
   ParseJSON off the critical path. */
if (!g_authOK && String(g_code) != "") {
  try {
    var body = '{"grant_type":"authorization_code"';
    body += ',"code":"'          + jsonSafe(g_code)    + '"';
    body += ',"client_id":"'     + jsonSafe(g_clientId)+ '"';
    body += ',"client_secret":"' + jsonSafe(g_secret)  + '"';
    body += ',"redirect_uri":"'  + jsonSafe(g_pageUrl) + '"}';

    var tokenUrl = "https://" + g_tenant + ".auth.marketingcloudapis.com/v2/token";
    var tokenRes = HTTP.Post(tokenUrl, "application/json", body);

    if (tokenRes != null && tokenRes.StatusCode == 200) {
      g_authOK = true;
      if (tokenRes.Response != null && tokenRes.Response.length > 0) {
        g_tokenRaw = String(tokenRes.Response[0]);
      }
      g_dbg = g_dbg + " | token=200";
    } else {
      g_dbg = g_dbg + " | token=" + (tokenRes == null ? "null" : String(tokenRes.StatusCode));
      if (tokenRes != null && tokenRes.Response != null && tokenRes.Response.length > 0) {
        g_dbg = g_dbg + " | body=" + String(tokenRes.Response[0]);
      }
    }
  } catch (tokenEx) {
    g_authOK = false;
    g_dbg = g_dbg + " | token threw: " + (tokenEx.message != null ? tokenEx.message : String(tokenEx));
  }
}

/* ---- TRY 2 of 2: who is the logged-in Marketing Cloud user? ---------------
   Purely cosmetic - it pre-fills the email box so the user does not have to
   type it. If any part of this fails the page carries on exactly as before.
   Sequential with TRY 1, never nested inside it. */
if (g_authOK && g_tokenRaw != "") {
  try {
    var tj = Platform.Function.ParseJSON(g_tokenRaw);
    var at = (tj != null) ? tj.access_token : null;
    g_dbg = g_dbg + " | tokenParsed=" + (tj == null ? "null" : "yes")
                  + " | accessToken=" + ((at == null || String(at) == "") ? "MISSING" : "present");

    if (at != null && String(at) != "") {
      var uiUrl = "https://" + g_tenant + ".auth.marketingcloudapis.com/v2/userinfo";
      var ui = HTTP.Get(uiUrl, ["Authorization"], ["Bearer " + String(at)]);

      /* v12: DO NOT gate this on ui.StatusCode.

         v11 did, and the GATE line came back "userinfo=undefined" - meaning the
         object came back fine but simply has no StatusCode property. The three
         argument HTTP.Get returns a different shape from HTTP.Post, and the
         article never checks a status here either; it reads ["Content"] and
         parses it. That check was my addition and it was silently throwing away
         a perfectly good response. Read the body, try each shape in turn. */
      var uiBody = "";
      if (ui != null) {
        if (ui["Content"] != null && String(ui["Content"]) != "") {
          uiBody = String(ui["Content"]);
        } else if (ui.Response != null && ui.Response.length > 0) {
          uiBody = String(ui.Response[0]);
        }
      }
      g_dbg = g_dbg + " | uiBodyLen=" + uiBody.length;

      if (uiBody != "") {
        var uj = Platform.Function.ParseJSON(uiBody);
        g_dbg = g_dbg + " | parsed=" + (uj == null ? "null" : "yes")
                      + " | userNode=" + ((uj != null && uj.user != null) ? "yes" : "no");

        if (uj != null && uj.user != null) {
          if (uj.user.email != null && String(uj.user.email) != "") {
            g_mcEmail = String(uj.user.email);
          } else if (uj.user.preferred_username != null) {
            g_mcEmail = String(uj.user.preferred_username);
          }
        }
        g_dbg = g_dbg + " | mcEmail=[" + g_mcEmail + "]";
      }
    }
  } catch (userEx) {
    /* Non-fatal on purpose. No pre-fill, nothing else changes. */
    g_dbg = g_dbg + " | userinfo THREW: "
          + (userEx.message != null ? userEx.message : String(userEx));
  }
}

Variable.SetValue("@authOK",      g_authOK ? "true" : "false");
Variable.SetValue("@mcUserEmail", g_mcEmail);
Variable.SetValue("@authDebug",   g_dbg);
</script>

/* ============================================================================
   BLOCK 0c — ACT ON THE VERDICT

   Three outcomes, and note that NONE of them can produce a redirect loop:

     PASSED, and we got here on a one-time ?code=
        -> bounce once more to a clean ?sessionkey= URL. This gets the
           single-use code out of the address bar, so pressing F5 later
           re-uses the session key instead of a burnt code.

     PASSED on a session key
        -> fall straight through into the real page. No redirect.

     FAILED
        -> if there was no code at all, send them to the login screen.
           If there WAS a code and it still failed, the credentials or the
           redirect URI are wrong; redirecting again would loop forever, so
           show a dead-end message instead.
   ============================================================================ */

  /* v9: THE CLEAN-URL REDIRECT THAT USED TO BE HERE HAS BEEN DELETED.

     It ran after a successful code validation and bounced to ?sessionkey=...
     purely to tidy the address bar. It was my addition - the article does not
     do this, it renders straight after validating the code - and it is what
     produced ERR_TOO_MANY_REDIRECTS:

         page -> /v2/authorize -> ?code= -> code OK -> redirect to ?sessionkey=
              -> session key fails -> no code -> /v2/authorize -> forever

     With it gone there is exactly ONE Redirect() left that can fire on any
     given page load, and a load carrying a ?code= ALWAYS renders or stops.
     A redirect loop is now structurally impossible, whatever else is wrong.

     Cost: the URL keeps ?code= after signing in, and auth codes are single
     use, so pressing F5 on that URL shows the "sign in again" message. One
     click. That is a fair trade for a page that cannot loop. */

  IF @authOK != "true" THEN

    IF @code != "" THEN
      SET @authFatal = "true"
    ELSEIF @authDebugMode != "true" THEN
      Redirect(@authURL)
    ELSE
      SET @authFatal = "true"
    ENDIF

  ENDIF

/* ==================== v14: DATA ACCESS REVERTED TO THE KNOWN-GOOD FORM =====

   v13 replaced Platform.Function.InsertData / UpdateData / Lookup / LookupRows
   with DataExtension.Retrieve + DataExtension.Init(CustomerKey), plus runtime
   column introspection and a read-back check. That was a theory about why the
   writes were failing, and the theory was wrong - the "ENT." names resolve
   perfectly well in this BU, and the Platform.Function.* calls work.

   So the whole data-access layer below is now IDENTICAL to the version that is
   confirmed working. No DataExtension.Init, no CustomerKey resolution, no
   Fields.Retrieve, no read-back.

   Everything else on the page is untouched v13: the OAuth gate in BLOCK 0, the
   /v2/userinfo welcome screen, the single-read request rule, the diagnostics
   switch, and rendering the success screen directly from the POST response
   rather than doing a Post/Redirect/Get.

   WHY THE SUCCESS REDIRECT IS STILL NOT BACK

/* Plain-JScript trim.
   Platform.Function.Trim() throws the security descriptor error in this BU,
   AND this engine (Jint) throws "StartIndex cannot be less than zero" on
   String.substring(). So this builds the result one character at a time —
   no substring, no slice, no regex. */
function isWs(c) {
  return (c == " " || c == "\t" || c == "\r" || c == "\n");
}

function trimStr(v) {
  if (v == null) { return ""; }
  var s = String(v);
  if (s.length == 0) { return ""; }
  var out = "";
  var pending = "";
  var started = false;
  for (var i = 0; i < s.length; i++) {
    var c = s.charAt(i);
    if (isWs(c)) {
      if (started) { pending = pending + c; }
    } else {
      if (pending != "") { out = out + pending; pending = ""; }
      out = out + c;
      started = true;
    }
  }
  return out;
}

/* Truncate without substring, for the 4000-char ErrorMessage column. */
function cut(v, maxLen) {
  if (v == null) { return ""; }
  var s = String(v);
  if (s.length <= maxLen) { return s; }
  var out = "";
  for (var i = 0; i < maxLen; i++) { out = out + s.charAt(i); }
  return out;
}

/* Read an AMPscript variable as a trimmed string. This replaces every
   Request.GetFormField / GetQueryStringParameter call the page used to make. */
function av(name) {
  var v = Variable.GetValue(name);
  if (v == null) { return ""; }
  return trimStr(v);
}

/* Read an AMPscript variable that holds a number. */
function avNum(name) {
  var v = Variable.GetValue(name);
  if (v == null || v == "") { return 0; }
  var n = parseInt(String(v), 10);
  if (isNaN(n)) { return 0; }
  return n;
}

function len(v) {
  if (v == null) { return 0; }
  return String(v).length;
}

/* Escape values before printing them back into HTML. split/join, not regex. */
function esc(v) {
  if (v == null) { return ""; }
  var s = String(v);
  s = s.split("&").join("&amp;");
  s = s.split("<").join("&lt;");
  s = s.split(">").join("&gt;");
  s = s.split('"').join("&quot;");
  return s;
}

/* Describe an exception without Platform.Function.Stringify. */
function errText(ex) {
  if (ex == null) { return "unknown error"; }
  var m = "";
  if (ex.message != null)     { m = m + ex.message; }
  if (ex.description != null) { m = m + " | " + ex.description; }
  if (m == "")                { m = String(ex); }
  return m;
}

<script>
  /* Browser-side JavaScript — handles instant screen switching. */

  function showScreen(name, stepperStep) {
    var screens = document.getElementsByClassName('screen');
    for (var i = 0; i < screens.length; i++) {
      screens[i].className = screens[i].className.replace(' active', '');
    }
    var target = document.getElementById('screen-' + name);
    if (target) { target.className += ' active'; }

    var stepper = document.getElementById('stepper');
    stepper.setAttribute('data-active', stepperStep);
    stepper.style.display = (name === 'done') ? 'none' : 'flex';
  }

  /* v8: put the typed address into the "Signed in as" pills on every screen. */
  function setPillEmail(v) {
    var els = document.getElementsByClassName('pill-email');
    for (var i = 0; i < els.length; i++) {
      while (els[i].firstChild) { els[i].removeChild(els[i].firstChild); }
      els[i].appendChild(document.createTextNode(v));
    }
  }

  /* v8: replaces the old server round trip. Shape check only - deliberately
     NOT an identity check. Identity was already settled by the OAuth gate. */
  /* v10: "1" when the address came from /v2/userinfo rather than a text box. */
  var haveMcIdentity = ("%%=v(@hasMcIdentity)=%%" === "1");

  function goToChoose() {
    var box = document.getElementById('txtEmailAddress');
    var err = document.getElementById('emailInlineError');

    /* Nothing to validate - Marketing Cloud told us who this is. */
    if (haveMcIdentity) {
      setPillEmail(box.value);
      showScreen('choose', '2');
      return;
    }

    var v = box.value.replace(/^\s+/, '').replace(/\s+$/, '');
    var at = v.indexOf('@');
    var dot = v.lastIndexOf('.');

    if (v.length === 0 || at < 1 || dot < at + 2 || dot === v.length - 1) {
      err.innerHTML = 'Please enter a valid email address.';
      err.style.display = 'block';
      box.focus();
      return;
    }

    err.style.display = 'none';
    box.value = v;
    setPillEmail(v);
    showScreen('choose', '2');
  }

  function chooseMode(mode) {
    document.getElementById('ddlMode').value = mode;
    showScreen(mode, '3');
  }

  function backToChoose() {
    document.getElementById('ddlMode').value = '';
    showScreen('choose', '2');
  }

  function startOver() {
    /* v6: keep the session key. Stripping the query string entirely would
       drop it and force a pointless round trip through the login server. */
    window.location.href = "%%=v(@PageURL)=%%?sessionkey=%%=v(@sessionKeyUrl)=%%";
  }

  /* Decide which screen to show, based on what the server logic worked out. */
  var serverStep = "%%=v(@renderStep)=%%";

  /* v8: keep the pills in step with whatever is in the box on a server render. */
  var emailBox = document.getElementById('txtEmailAddress');
  if (emailBox && emailBox.value !== "") { setPillEmail(emailBox.value); }

  if (serverStep === "choose") {
    showScreen('choose', '2');
  } else if (serverStep === "add") {
    document.getElementById('ddlMode').value = 'add';
    showScreen('add', '3');
  } else if (serverStep === "update") {
    document.getElementById('ddlMode').value = 'update';
    showScreen('update', '3');
  } else if (serverStep === "done") {
    showScreen('done', '3');
  } else {
    showScreen('email', '1');
  }
</script>
